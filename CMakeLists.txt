cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/vcpkg/scripts/buildsystems/vcpkg.cmake")
    set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_SOURCE_DIR}/tools/vcpkg/scripts/buildsystems/vcpkg.cmake" CACHE FILEPATH "vcpkg toolchain")
endif()
project(Sayra VERSION 0.1.0 LANGUAGES CXX)
find_package(CURL CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(unofficial-sqlite3 CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_library(sayra-core
    src/types.cpp src/config.cpp src/events.cpp src/transport.cpp src/providers.cpp
    src/embeddings.cpp src/routing.cpp src/projects.cpp src/memory.cpp
    src/tools.cpp src/orchestrator.cpp src/realtime.cpp)
target_include_directories(sayra-core PUBLIC include)
target_compile_features(sayra-core PUBLIC cxx_std_20)
set_target_properties(sayra-core PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(sayra-core PUBLIC nlohmann_json::nlohmann_json Threads::Threads PRIVATE CURL::libcurl unofficial::sqlite3::sqlite3)
target_compile_definitions(sayra-core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
if(WIN32)
    target_compile_definitions(sayra-core PUBLIC _WIN32_WINNT=0x0A00 WINVER=0x0A00)
endif()
if(MSVC)
    target_compile_options(sayra-core PUBLIC /utf-8 PRIVATE /W4 /permissive-)
endif()
add_executable(sayra-cli apps/sayra-cli/main.cpp)
target_link_libraries(sayra-cli PRIVATE sayra-core)
option(SAYRA_BUILD_MODEL_DEMO "Keep original university model diagnostics" ON)
if(SAYRA_BUILD_MODEL_DEMO)
    add_executable(sayra-model-tests Sayra.cpp)
    target_compile_features(sayra-model-tests PRIVATE cxx_std_20)
    target_compile_definitions(sayra-model-tests PRIVATE NOMINMAX)
    target_link_libraries(sayra-model-tests PRIVATE CURL::libcurl nlohmann_json::nlohmann_json)
    if(MSVC)
        target_compile_options(sayra-model-tests PRIVATE /utf-8)
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(sayra-core-tests tests/core_tests.cpp)
    target_link_libraries(sayra-core-tests PRIVATE sayra-core)
    add_test(NAME sayra-core-tests COMMAND sayra-core-tests)
endif()
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT sayra-cli)
